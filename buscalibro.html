function evaluarTexto(textoLimpio) { // 1. Limpieza base: Todo mayúsculas, unificar espacios, quitar símbolos raros let texto = textoLimpio.toUpperCase() .replace(/[^A-Z0-9.\- ]/g, ' ') // Solo letras, números, puntos y guiones .replace(/\s+/g, ' ') .trim(); if (!texto || texto.length < 3) return; // 2. Extraer inteligentemente desde el primer número // (Ignora carteles que digan "Sala", "Estante", etc.) let matchIndex = texto.search(/\d/); if (matchIndex === -1) return; // Si no hay números, no es una signatura de biblioteca let signaturaLeida = texto.substring(matchIndex); // Mostrar en pantalla lo que el sistema "entiende" que está leyendo textoVisto.innerText = signaturaLeida; // 3. Preparar cadenas comprimidas (sin espacios) para evitar fallos del OCR // Ejemplo: "821.134 ALV" se convierte en "821.134ALV" let leidoStr = signaturaLeida.replace(/\s+/g, ''); let targetStr = sigOriginal.replace(/\s+/g, ''); let baseStr = sigBuscada.replace(/\s+/g, ''); // 4. Verificaciones de Éxito if (leidoStr.includes(targetStr)) { estadoActual = "ENCONTRADO"; actualizarPanel("🎯", "¡LIBRO ENCONTRADO!", sigOriginal, "dir-ok"); notificarCambio("ENCONTRADO", "Libro localizado. Puedes cogerlo."); return; } if (estadoActual === "BUSCANDO_ESTANTE" && leidoStr.includes(baseStr)) { estadoActual = "BUSCANDO_LIBRO"; actualizarPanel("📚", "Sección Correcta", "Ahora busca tu libro", "dir-arr"); notificarCambio("ESTANTE_ENCONTRADO", "Sección correcta."); return; } // 5. Comparación Direccional Continua // Se usa localeCompare estándar. En la clasificación decimal (CDU), // 821.2 va después de 821.134 porque el caracter '2' es mayor que '1'. let comp = leidoStr.localeCompare(targetStr); // Ignoramos lecturas demasiado cortas para evitar "saltos" en las flechas if (leidoStr.length < 4) return; if (comp < 0) { // Lo que enfocamos es MENOR (ej. Leemos ALV y buscamos CER) -> Está a la Derecha actualizarPanel("▶", "DERECHA", `Viendo: ${signaturaLeida}`, "dir-der"); notificarCambio("DERECHA", "Derecha"); } else if (comp > 0) { // Lo que enfocamos es MAYOR (ej. Leemos GOM y buscamos CER) -> Está a la Izquierda actualizarPanel("◀", "IZQUIERDA", `Viendo: ${signaturaLeida}`, "dir-izq"); notificarCambio("IZQUIERDA", "Izquierda"); } }
